The same thing from a terminal

# public, no credential
curl -s localhost:8080/projects | grep -o 'Glass Signal'

# a judge reading their own ballots
curl -s -H 'Cookie: df_session=jdg_a_91bc' localhost:8080/api/judge/scores

# the other judge trying the same url: 403, and an audit entry
curl -s -o /dev/null -w '%{http_code}\n' \
  -H 'Cookie: df_session=jdg_b_44de' 'localhost:8080/api/judge/scores?judge=jdg_01'

# the organizer on that same url: 200, because they hold peer_scores
curl -s -o /dev/null -w '%{http_code}\n' \
  -H 'Cookie: df_session=org_7f2a' 'localhost:8080/api/judge/scores?judge=jdg_01'
